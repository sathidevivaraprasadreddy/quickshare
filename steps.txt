-- QuickShare Database Schema for MySQL / TiDB / MariaDB
CREATE DATABASE IF NOT EXISTS quickshare_db
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;

USE quickshare_db;

-- Table: share_sessions
-- Represents a single upload session (which can contain one or multiple files)
CREATE TABLE IF NOT EXISTS share_sessions (
  id INT AUTO_INCREMENT PRIMARY KEY,
  share_token VARCHAR(64) NOT NULL UNIQUE,
  revoke_token VARCHAR(64) NOT NULL,
  total_size_bytes BIGINT NOT NULL DEFAULT 0,
  file_count INT NOT NULL DEFAULT 1,
  expires_at DATETIME NOT NULL,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  is_revoked TINYINT(1) NOT NULL DEFAULT 0,
  download_count INT NOT NULL DEFAULT 0,
  INDEX idx_share_token (share_token),
  INDEX idx_expires_at (expires_at),
  INDEX idx_is_revoked (is_revoked)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Table: share_files
-- Represents individual files uploaded in a session
CREATE TABLE IF NOT EXISTS share_files (
  id INT AUTO_INCREMENT PRIMARY KEY,
  session_id INT NOT NULL,
  original_name VARCHAR(255) NOT NULL,
  stored_name VARCHAR(255) NOT NULL,
  file_path VARCHAR(500) NOT NULL,
  file_size BIGINT NOT NULL,
  mime_type VARCHAR(127) NOT NULL DEFAULT 'application/octet-stream',
  storage_driver VARCHAR(32) NOT NULL DEFAULT 'local',
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  INDEX idx_session_id (session_id),
  CONSTRAINT fk_session_files FOREIGN KEY (session_id) 
    REFERENCES share_sessions (id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

import fs from 'fs';
import path from 'path';
import { fileURLToPath } from 'url';
import mysql from 'mysql2/promise';
import dotenv from 'dotenv';

dotenv.config();

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

async function runMigration() {
  const isUrlConfig = Boolean(process.env.DATABASE_URL);
  let connection;

  try {
    if (isUrlConfig) {
      connection = await mysql.createConnection({
        uri: process.env.DATABASE_URL,
        multipleStatements: true,
        ssl: process.env.DATABASE_URL.includes('ssl=') ? undefined : { rejectUnauthorized: false }
      });
      console.log('Connected using DATABASE_URL.');
    } else {
      connection = await mysql.createConnection({
        host: process.env.DB_HOST || '127.0.0.1',
        port: Number(process.env.DB_PORT) || 3306,
        user: process.env.DB_USER || 'root',
        password: process.env.DB_PASSWORD || '',
        multipleStatements: true
      });
      console.log(`Connected to MySQL server at ${process.env.DB_HOST || '127.0.0.1'}:${process.env.DB_PORT || 3306}`);
    }

    const schemaPath = path.join(__dirname, 'schema.sql');
    const sql = fs.readFileSync(schemaPath, 'utf8');

    console.log('Executing database migrations...');
    await connection.query(sql);
    console.log('Migration completed successfully.');

    await connection.end();
    process.exit(0);
  } catch (err) {
    console.error('Migration failed:', err);
    if (connection) await connection.end();
    process.exit(1);
  }
}

runMigration();

import mysql from 'mysql2/promise';
import dotenv from 'dotenv';

dotenv.config();

let pool;

export function getDbPool() {
  if (pool) return pool;

  const isUrlConfig = Boolean(process.env.DATABASE_URL);

  if (isUrlConfig) {
    pool = mysql.createPool({
      uri: process.env.DATABASE_URL,
      waitForConnections: true,
      connectionLimit: 10,
      queueLimit: 0,
      ssl: process.env.DATABASE_URL.includes('ssl=') ? undefined : { rejectUnauthorized: false }
    });
  } else {
    pool = mysql.createPool({
      host: process.env.DB_HOST || '127.0.0.1',
      port: Number(process.env.DB_PORT) || 3306,
      user: process.env.DB_USER || 'root',
      password: process.env.DB_PASSWORD || '',
      database: process.env.DB_NAME || 'quickshare_db',
      waitForConnections: true,
      connectionLimit: 10,
      queueLimit: 0
    });
  }

  return pool;
}

export async function testDbConnection() {
  try {
    const db = getDbPool();
    const [rows] = await db.query('SELECT 1 + 1 AS solution');
    return { connected: true, solution: rows[0].solution };
  } catch (error) {
    return { connected: false, error: error.message };
  }
}

export default getDbPool();

import crypto from 'crypto';
import path from 'path';

/**
 * Generate a cryptographically secure, unpredictable token for public sharing.
 * 32 characters (16 bytes hex) gives 2^128 possible values.
 */
export function generateShareToken() {
  return crypto.randomBytes(16).toString('hex');
}

/**
 * Generate a cryptographically secure token for revoking/deleting a session.
 * 48 characters (24 bytes hex) ensures high entropy.
 */
export function generateRevokeToken() {
  return crypto.randomBytes(24).toString('hex');
}

/**
 * Sanitize original filename to prevent path traversal and shell injection risks.
 * Preserves normal characters, dashes, underscores, dots, and spaces.
 */
export function sanitizeFilename(originalName) {
  if (!originalName || typeof originalName !== 'string') {
    return 'unnamed_file';
  }

  // Strip path traversal attempts and directory components
  const baseName = path.basename(originalName.trim());

  // Replace null bytes and dangerous control characters
  const cleanName = baseName
    .replace(/[\x00-\x1f\x80-\x9f]/g, '')
    .replace(/[<>:"/\\|?*]/g, '_');

  return cleanName.length > 0 ? cleanName.substring(0, 200) : 'file';
}

/**
 * Generate an unpredictable internal stored filename.
 * Uses 16 random bytes + timestamp + sanitized extension.
 * This guarantees no collisions and zero correlation with original filesystem names.
 */
export function generateStoredFilename(originalName) {
  const ext = path.extname(originalName || '').toLowerCase().substring(0, 10);
  const safeExt = ext.replace(/[^a-z0-9.]/gi, '');
  const randomPart = crypto.randomBytes(16).toString('hex');
  const timestamp = Date.now();
  return `${timestamp}_${randomPart}${safeExt}`;
}

/**
 * Format bytes into human-readable units (e.g. 1.25 MB)
 */
export function formatBytes(bytes, decimals = 2) {
  if (bytes === 0) return '0 Bytes';
  const k = 1024;
  const dm = decimals < 0 ? 0 : decimals;
  const sizes = ['Bytes', 'KB', 'MB', 'GB', 'TB'];
  const i = Math.floor(Math.log(bytes) / Math.log(k));
  return parseFloat((bytes / Math.pow(k, i)).toFixed(dm)) + ' ' + sizes[i];
}

import fs from 'fs';
import path from 'path';
import { fileURLToPath } from 'url';
import {
  S3Client,
  PutObjectCommand,
  GetObjectCommand,
  DeleteObjectCommand,
  DeleteObjectsCommand,
  HeadObjectCommand
} from '@aws-sdk/client-s3';

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

const DEFAULT_UPLOAD_DIR = path.resolve(__dirname, '../../uploads');
const UPLOAD_DIR = process.env.UPLOAD_DIR
  ? path.resolve(process.cwd(), process.env.UPLOAD_DIR)
  : DEFAULT_UPLOAD_DIR;

// Ensure upload directory exists for local driver
if (!fs.existsSync(UPLOAD_DIR)) {
  fs.mkdirSync(UPLOAD_DIR, { recursive: true });
}

export class StorageService {
  constructor() {
    this.driver = (process.env.STORAGE_DRIVER || 'local').toLowerCase();
    this.uploadDir = UPLOAD_DIR;
    this.s3Bucket = process.env.S3_BUCKET_NAME || 'quickshare-files';

    if (this.driver === 's3') {
      const endpoint = process.env.S3_ENDPOINT || undefined;
      const region = process.env.S3_REGION || 'auto';
      const accessKeyId = process.env.S3_ACCESS_KEY_ID;
      const secretAccessKey = process.env.S3_SECRET_ACCESS_KEY;

      if (!accessKeyId || !secretAccessKey) {
        console.warn('⚠️ S3 credentials missing. Falling back to local storage.');
        this.driver = 'local';
      } else {
        this.s3Client = new S3Client({
          region,
          endpoint,
          credentials: {
            accessKeyId,
            secretAccessKey
          },
          forcePathStyle: true // Needed for Supabase Storage & Cloudflare R2
        });
        console.log(`[Storage] Initialized S3-compatible driver (Bucket: ${this.s3Bucket})`);
      }
    }
  }

  getUploadDir() {
    return this.uploadDir;
  }

  /**
   * Verify and resolve absolute path safely to prevent directory traversal in local driver.
   */
  resolveSafePath(storedName) {
    const sanitized = path.basename(storedName);
    const fullPath = path.resolve(this.uploadDir, sanitized);

    if (!fullPath.startsWith(this.uploadDir)) {
      throw new Error('Access denied: Invalid file path traversal attempt.');
    }
    return fullPath;
  }

  /**
   * Check if file exists
   */
  async fileExists(storedName) {
    if (this.driver === 's3') {
      try {
        await this.s3Client.send(
          new HeadObjectCommand({
            Bucket: this.s3Bucket,
            Key: storedName
          })
        );
        return true;
      } catch (err) {
        return false;
      }
    }

    try {
      const fullPath = this.resolveSafePath(storedName);
      return fs.existsSync(fullPath);
    } catch {
      return false;
    }
  }

  /**
   * Upload file to remote storage (called during upload if driver is s3)
   */
  async persistUploadedFile(storedName, originalMimeType) {
    if (this.driver === 's3') {
      const fullPath = this.resolveSafePath(storedName);
      const fileStream = fs.createReadStream(fullPath);

      await this.s3Client.send(
        new PutObjectCommand({
          Bucket: this.s3Bucket,
          Key: storedName,
          Body: fileStream,
          ContentType: originalMimeType || 'application/octet-stream'
        })
      );

      // Clean up local temp file after uploading to S3
      await fs.promises.unlink(fullPath).catch(() => {});
    }
  }

  /**
   * Get readable stream for downloading
   */
  async getFileStream(storedName) {
    if (this.driver === 's3') {
      const command = new GetObjectCommand({
        Bucket: this.s3Bucket,
        Key: storedName
      });
      const response = await this.s3Client.send(command);
      return response.Body; // Node stream
    }

    const fullPath = this.resolveSafePath(storedName);
    if (!fs.existsSync(fullPath)) {
      throw new Error('File not found on storage.');
    }
    return fs.createReadStream(fullPath);
  }

  /**
   * Delete a single stored file
   */
  async deleteFile(storedName) {
    if (this.driver === 's3') {
      try {
        await this.s3Client.send(
          new DeleteObjectCommand({
            Bucket: this.s3Bucket,
            Key: storedName
          })
        );
        return true;
      } catch (err) {
        console.error(`[S3 Delete Error] ${storedName}:`, err.message);
        return false;
      }
    }

    try {
      const fullPath = this.resolveSafePath(storedName);
      if (fs.existsSync(fullPath)) {
        await fs.promises.unlink(fullPath);
        return true;
      }
      return false;
    } catch (err) {
      console.error(`Error deleting file ${storedName}:`, err.message);
      return false;
    }
  }

  /**
   * Delete multiple stored files
   */
  async deleteFiles(storedNames) {
    if (storedNames.length === 0) return 0;

    if (this.driver === 's3') {
      try {
        await this.s3Client.send(
          new DeleteObjectsCommand({
            Bucket: this.s3Bucket,
            Delete: {
              Objects: storedNames.map((Key) => ({ Key }))
            }
          })
        );
        return storedNames.length;
      } catch (err) {
        console.error('[S3 Batch Delete Error]:', err.message);
        return 0;
      }
    }

    const results = await Promise.all(
      storedNames.map((name) => this.deleteFile(name))
    );
    return results.filter(Boolean).length;
  }
}

export const storageService = new StorageService();
export default storageService;

import cron from 'node-cron';
import { getDbPool } from '../config/db.js';
import storageService from './storageService.js';

/**
 * Scan database for expired sessions, delete associated files from disk/storage,
 * and remove the expired sessions from the database.
 */
export async function cleanupExpiredFiles() {
  const db = getDbPool();
  try {
    // 1. Find all sessions that are expired (expires_at < NOW()) or marked revoked
    // and retrieve their stored file paths
    const [expiredFiles] = await db.query(
      `SELECT f.id AS file_id, f.stored_name, s.id AS session_id, s.share_token 
       FROM share_files f
       JOIN share_sessions s ON f.session_id = s.id
       WHERE s.expires_at < NOW() OR s.is_revoked = 1`
    );

    if (expiredFiles.length === 0) {
      return { cleanedFilesCount: 0, cleanedSessionsCount: 0 };
    }

    // 2. Delete physical files from storage
    const storedNames = expiredFiles.map((f) => f.stored_name);
    await storageService.deleteFiles(storedNames);

    // 3. Delete expired sessions from database (CASCADE deletes file records)
    const sessionIds = [...new Set(expiredFiles.map((f) => f.session_id))];
    const placeholders = sessionIds.map(() => '?').join(',');

    const [deleteResult] = await db.query(
      `DELETE FROM share_sessions WHERE id IN (${placeholders})`,
      sessionIds
    );

    console.log(
      `[Cleanup] Removed ${deleteResult.affectedRows} expired session(s) and ${storedNames.length} file(s).`
    );

    return {
      cleanedFilesCount: storedNames.length,
      cleanedSessionsCount: deleteResult.affectedRows
    };
  } catch (err) {
    console.error('[Cleanup] Error during expired files cleanup:', err.message);
    return { error: err.message };
  }
}

/**
 * Start the background cron job for automatic cleanup.
 */
export function startCleanupJob() {
  const schedule = process.env.CLEANUP_CRON_SCHEDULE || '*/15 * * * *';
  console.log(`[Cleanup] Background cleaner scheduled with pattern: ${schedule}`);

  cron.schedule(schedule, async () => {
    console.log('[Cleanup] Running scheduled expired files cleaner...');
    await cleanupExpiredFiles();
  });

  // Also run once shortly after server boot to clean anything that expired while offline
  setTimeout(() => {
    cleanupExpiredFiles().catch((err) =>
      console.error('[Cleanup] Initial startup cleanup error:', err.message)
    );
  }, 5000);
}

const MAX_FILES_COUNT = Number(process.env.MAX_FILES_COUNT) || 10;

// Block dangerous executable extensions that could harm a server or recipient if executed
const BLOCKED_EXTENSIONS = new Set([
  '.exe', '.bat', '.cmd', '.sh', '.bash', '.vbs', '.msi', '.com', 
  '.scr', '.pif', '.application', '.gadget', '.hta', '.cpl', '.msc', 
  '.jar', '.php', '.asp', '.aspx', '.jsp', '.cgi', '.pl'
]);

const storage = multer.diskStorage({
  destination: (req, file, cb) => {
    cb(null, storageService.getUploadDir());
  },
  filename: (req, file, cb) => {
    // Decode original name if latin1 encoded by multer
    let originalName = file.originalname;
    try {
      originalName = Buffer.from(file.originalname, 'latin1').toString('utf8');
    } catch {
      originalName = file.originalname;
    }
    file.sanitizedOriginalName = sanitizeFilename(originalName);
    const storedName = generateStoredFilename(file.sanitizedOriginalName);
    cb(null, storedName);
  }
});

const fileFilter = (req, file, cb) => {
  const ext = path.extname(file.originalname || '').toLowerCase();
  if (BLOCKED_EXTENSIONS.has(ext)) {
    return cb(
      new Error(`Files with extension '${ext}' are restricted for security reasons.`),
      false
    );
  }
  cb(null, true);
};

export const upload = multer({
  storage,
  fileFilter,
  limits: {
    fileSize: MAX_FILE_SIZE_MB * 1024 * 1024, // bytes
    files: MAX_FILES_COUNT
  }
});

import multer from 'multer';

export function errorHandler(err, req, res, next) {
  console.error('[Error Handler]', err);

  // Multer errors
  if (err instanceof multer.MulterError) {
    if (err.code === 'LIMIT_FILE_SIZE') {
      const maxMb = process.env.MAX_FILE_SIZE_MB || 50;
      return res.status(400).json({
        success: false,
        error: `File too large. Maximum allowed size is ${maxMb}MB per file.`
      });
    }
    if (err.code === 'LIMIT_FILE_COUNT') {
      const maxCount = process.env.MAX_FILES_COUNT || 10;
      return res.status(400).json({
        success: false,
        error: `Too many files. Maximum allowed is ${maxCount} files per upload session.`
      });
    }
    return res.status(400).json({
      success: false,
      error: `Upload error: ${err.message}`
    });
  }

  // Custom validation or file filter error
  if (err.message && err.message.includes('restricted for security reasons')) {
    return res.status(400).json({
      success: false,
      error: err.message
    });
  }

  // Path traversal or illegal path errors
  if (err.message && err.message.includes('Access denied')) {
    return res.status(403).json({
      success: false,
      error: err.message
    });
  }

  // Default server error
  const statusCode = err.statusCode || 500;
  res.status(statusCode).json({
    success: false,
    error: err.message || 'An unexpected internal server error occurred.'
  });
}

import path from 'path';
import archiver from 'archiver';
import mime from 'mime-types';
import { getDbPool } from '../config/db.js';
import storageService from '../services/storageService.js';
import {
  generateShareToken,
  generateRevokeToken,
  sanitizeFilename
} from '../utils/fileSecurity.js';

// Allowed expiration periods in hours
const ALLOWED_EXPIRY_HOURS = [1, 6, 24, 72, 168]; // 1 hour, 6 hours, 24 hours (1 day), 3 days, 7 days
const DEFAULT_EXPIRY_HOURS = 24;

/**
 * Create a new sharing session with one or multiple files
 */
export async function createShareSession(req, res, next) {
  const files = req.files;
  if (!files || files.length === 0) {
    return res.status(400).json({
      success: false,
      error: 'Please select at least one file to upload.'
    });
  }

  // Parse and validate expiry
  let expiryHours = parseInt(req.body.expiryHours, 10);
  if (!ALLOWED_EXPIRY_HOURS.includes(expiryHours)) {
    expiryHours = DEFAULT_EXPIRY_HOURS;
  }

  const expiresAt = new Date(Date.now() + expiryHours * 60 * 60 * 1000);
  const shareToken = generateShareToken();
  const revokeToken = generateRevokeToken();

  let totalSizeBytes = 0;
  for (const f of files) {
    totalSizeBytes += f.size;
  }

  const db = getDbPool();
  const connection = await db.getConnection();

  try {
    await connection.beginTransaction();

    // 1. Insert session record
    const [sessionResult] = await connection.query(
      `INSERT INTO share_sessions 
       (share_token, revoke_token, total_size_bytes, file_count, expires_at)
       VALUES (?, ?, ?, ?, ?)`,
      [shareToken, revokeToken, totalSizeBytes, files.length, expiresAt]
    );

    const sessionId = sessionResult.insertId;

    // 2. Insert file records
    for (const f of files) {
      const originalName = f.sanitizedOriginalName || sanitizeFilename(f.originalname);
      const storedName = f.filename;
      const mimeType = f.mimetype || mime.lookup(originalName) || 'application/octet-stream';

      await connection.query(
        `INSERT INTO share_files
         (session_id, original_name, stored_name, file_path, file_size, mime_type, storage_driver)
         VALUES (?, ?, ?, ?, ?, ?, ?)`,
        [
          sessionId,
          originalName,
          storedName,
          storedName, // relative safe name
          f.size,
          mimeType,
          storageService.driver
        ]
      );

      // Persist to remote S3 storage if configured
      await storageService.persistUploadedFile(storedName, mimeType);
    }

    await connection.commit();

    return res.status(201).json({
      success: true,
      shareToken,
      revokeToken,
      expiresAt: expiresAt.toISOString(),
      fileCount: files.length,
      totalSizeBytes
    });
  } catch (err) {
    await connection.rollback();
    // Clean up uploaded files on disk if DB insertion failed
    for (const f of files) {
      if (f.filename) {
        await storageService.deleteFile(f.filename);
      }
    }
    next(err);
  } finally {
    connection.release();
  }
}

/**
 * Get share session metadata and file list by share token
 */
export async function getShareSession(req, res, next) {
  try {
    const { token } = req.params;
    if (!token || typeof token !== 'string') {
      return res.status(400).json({ success: false, error: 'Invalid share link.' });
    }

    const db = getDbPool();
    const [sessions] = await db.query(
      `SELECT id, share_token, total_size_bytes, file_count, expires_at, created_at, is_revoked, download_count
       FROM share_sessions
       WHERE share_token = ?`,
      [token]
    );

    if (sessions.length === 0) {
      return res.status(404).json({
        success: false,
        error: 'This sharing link does not exist or has already been deleted.'
      });
    }

    const session = sessions[0];

    // Check if session was revoked by owner
    if (session.is_revoked) {
      return res.status(410).json({
        success: false,
        isRevoked: true,
        error: 'This sharing session has been revoked by the owner.'
      });
    }

    // Check if session has expired
    const isExpired = new Date(session.expires_at) < new Date();
    if (isExpired) {
      return res.status(410).json({
        success: false,
        isExpired: true,
        error: 'This sharing link has expired.'
      });
    }

    // Get files for this session (never expose internal stored_name or file_path)
    const [files] = await db.query(
      `SELECT id, original_name, file_size, mime_type, created_at
       FROM share_files
       WHERE session_id = ?
       ORDER BY id ASC`,
      [session.id]
    );

    return res.json({
      success: true,
      session: {
        shareToken: session.share_token,
        totalSizeBytes: session.total_size_bytes,
        fileCount: session.file_count,
        expiresAt: session.expires_at,
        createdAt: session.created_at,
        downloadCount: session.download_count
      },
      files: files.map((f) => ({
        id: f.id,
        name: f.original_name,
        size: f.file_size,
        mimeType: f.mime_type,
        createdAt: f.created_at
      }))
    });
  } catch (err) {
    next(err);
  }
}

/**
 * Download a single file by file ID within a valid session
 */
export async function downloadSingleFile(req, res, next) {
  try {
    const { token, fileId } = req.params;
    const db = getDbPool();

    // 1. Verify session is active and valid
    const [sessions] = await db.query(
      `SELECT id, expires_at, is_revoked FROM share_sessions WHERE share_token = ?`,
      [token]
    );

    if (sessions.length === 0) {
      return res.status(404).json({ success: false, error: 'Sharing session not found.' });
    }

    const session = sessions[0];
    if (session.is_revoked) {
      return res.status(410).json({ success: false, error: 'Sharing session has been revoked.' });
    }
    if (new Date(session.expires_at) < new Date()) {
      return res.status(410).json({ success: false, error: 'Sharing link has expired.' });
    }

    // 2. Fetch file record
    const [files] = await db.query(
      `SELECT id, original_name, stored_name, file_size, mime_type
       FROM share_files
       WHERE id = ? AND session_id = ?`,
      [fileId, session.id]
    );

    if (files.length === 0) {
      return res.status(404).json({ success: false, error: 'Requested file not found in this session.' });
    }

    const file = files[0];

    // 3. Verify file exists on storage
    const exists = await storageService.fileExists(file.stored_name);
    if (!exists) {
      return res.status(404).json({ success: false, error: 'File data is missing from storage.' });
    }

    // 4. Update download count asynchronously
    db.query(`UPDATE share_sessions SET download_count = download_count + 1 WHERE id = ?`, [session.id]).catch(() => {});

    // 5. Send file with safe content disposition
    const encodedFilename = encodeURIComponent(file.original_name);
    res.setHeader(
      'Content-Disposition',
      `attachment; filename="${encodedFilename}"; filename*=UTF-8''${encodedFilename}`
    );
    res.setHeader('Content-Type', file.mime_type || 'application/octet-stream');
    res.setHeader('Content-Length', file.file_size);
    res.setHeader('X-Content-Type-Options', 'nosniff');
    res.setHeader('Cache-Control', 'no-cache, no-store, must-revalidate');

    const readStream = await storageService.getFileStream(file.stored_name);
    readStream.on('error', (streamErr) => {
      console.error('File stream error:', streamErr);
      if (!res.headersSent) {
        res.status(500).json({ success: false, error: 'Failed to stream file.' });
      }
    });

    readStream.pipe(res);
  } catch (err) {
    next(err);
  }
}

/**
 * Download all files in a session bundled into a zip archive
 */
export async function downloadAllFiles(req, res, next) {
  try {
    const { token } = req.params;
    const db = getDbPool();

    // Verify session
    const [sessions] = await db.query(
      `SELECT id, share_token, expires_at, is_revoked FROM share_sessions WHERE share_token = ?`,
      [token]
    );

    if (sessions.length === 0) {
      return res.status(404).json({ success: false, error: 'Sharing session not found.' });
    }

    const session = sessions[0];
    if (session.is_revoked) {
      return res.status(410).json({ success: false, error: 'Sharing session has been revoked.' });
    }
    if (new Date(session.expires_at) < new Date()) {
      return res.status(410).json({ success: false, error: 'Sharing link has expired.' });
    }

    // Fetch all files
    const [files] = await db.query(
      `SELECT id, original_name, stored_name, file_size
       FROM share_files
       WHERE session_id = ?`,
      [session.id]
    );

    if (files.length === 0) {
      return res.status(404).json({ success: false, error: 'No files found in this session.' });
    }

    // Update download count
    db.query(`UPDATE share_sessions SET download_count = download_count + 1 WHERE id = ?`, [session.id]).catch(() => {});

    const zipFilename = `QuickShare_${token.substring(0, 8)}.zip`;

    res.setHeader('Content-Disposition', `attachment; filename="${zipFilename}"`);
    res.setHeader('Content-Type', 'application/zip');
    res.setHeader('Cache-Control', 'no-cache, no-store, must-revalidate');

    const archive = archiver('zip', {
      zlib: { level: 5 } // balanced compression and speed
    });

    archive.on('error', (err) => {
      console.error('Archiver error:', err);
      if (!res.headersSent) {
        res.status(500).json({ success: false, error: 'Failed to create zip bundle.' });
      }
    });

    archive.pipe(res);

    // Keep track of names to prevent duplicate zip entries
    const seenNames = new Map();

    for (const f of files) {
      const exists = await storageService.fileExists(f.stored_name);
      if (exists) {
        let entryName = f.original_name;
        if (seenNames.has(entryName)) {
          const count = seenNames.get(entryName) + 1;
          seenNames.set(entryName, count);
          const ext = path.extname(entryName);
          const base = path.basename(entryName, ext);
          entryName = `${base} (${count})${ext}`;
        } else {
          seenNames.set(entryName, 0);
        }

        const fileStream = await storageService.getFileStream(f.stored_name);
        archive.append(fileStream, { name: entryName });
      }
    }

    await archive.finalize();
  } catch (err) {
    next(err);
  }
}

/**
 * Revoke/delete a share session early using the secret revokeToken
 */
export async function revokeShareSession(req, res, next) {
  try {
    const { shareToken, revokeToken } = req.body;
    if (!shareToken || !revokeToken) {
      return res.status(400).json({
        success: false,
        error: 'Both shareToken and revokeToken are required to revoke a session.'
      });
    }

    const db = getDbPool();

    // Verify tokens match
    const [sessions] = await db.query(
      `SELECT id, is_revoked FROM share_sessions WHERE share_token = ? AND revoke_token = ?`,
      [shareToken, revokeToken]
    );

    if (sessions.length === 0) {
      return res.status(403).json({
        success: false,
        error: 'Invalid authorization token or session does not exist.'
      });
    }

    const session = sessions[0];
    if (session.is_revoked) {
      return res.json({
        success: true,
        message: 'This sharing link was already revoked.'
      });
    }

    // Get files to delete from storage immediately
    const [files] = await db.query(
      `SELECT stored_name FROM share_files WHERE session_id = ?`,
      [session.id]
    );

    // Delete stored files
    const storedNames = files.map((f) => f.stored_name);
    await storageService.deleteFiles(storedNames);

    // Mark as revoked
    await db.query(
      `UPDATE share_sessions SET is_revoked = 1 WHERE id = ?`,
      [session.id]
    );

    return res.json({
      success: true,
      message: 'Sharing link has been revoked and files deleted successfully.'
    });
  } catch (err) {
    next(err);
  }
}

/**
 * Format bytes into human-readable unit
 */
export function formatFileSize(bytes) {
  if (bytes === 0 || !bytes) return '0 B';
  const k = 1024;
  const sizes = ['B', 'KB', 'MB', 'GB'];
  const i = Math.floor(Math.log(bytes) / Math.log(k));
  return `${parseFloat((bytes / Math.pow(k, i)).toFixed(1))} ${sizes[i]}`;
}

/**
 * Format remaining expiration time
 */
export function formatTimeRemaining(expiresAt) {
  if (!expiresAt) return '';
  const now = new Date().getTime();
  const target = new Date(expiresAt).getTime();
  const diff = target - now;

  if (diff <= 0) return 'Expired';

  const hours = Math.floor(diff / (1000 * 60 * 60));
  const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));

  if (hours >= 24) {
    const days = Math.floor(hours / 24);
    const remHours = hours % 24;
    return `${days}d ${remHours}h remaining`;
  }
  if (hours > 0) {
    return `${hours}h ${minutes}m remaining`;
  }
  return `${minutes}m remaining`;
}

/**
 * Upload multiple files with real-time progress callback
 */
export async function uploadFiles(files, expiryHours = 24, onProgress) {
  const formData = new FormData();
  formData.append('expiryHours', expiryHours);

  for (let i = 0; i < files.length; i++) {
    formData.append('files', files[i]);
  }

  const response = await api.post('/upload', formData, {
    headers: {
      'Content-Type': 'multipart/form-data'
    },
    onUploadProgress: (progressEvent) => {
      if (onProgress && progressEvent.total) {
        const percentCompleted = Math.round(
          (progressEvent.loaded * 100) / progressEvent.total
        );
        onProgress(percentCompleted, progressEvent.loaded, progressEvent.total);
      }
    }
  });

  return response.data;
}

/**
 * Fetch sharing session info and file listing
 */

  return response.data;
}

import React, { useRef, useState } from 'react';
import { UploadCloud, FolderPlus, AlertCircle, FilePlus2 } from 'lucide-react';

const MAX_FILE_SIZE_MB = 50;
const MAX_FILES_COUNT = 10;

export default function FileUploader({ onFilesSelected, onError }) {
  const [isDragging, setIsDragging] = useState(false);
  const fileInputRef = useRef(null);

  const handleDragOver = (e) => {
    e.preventDefault();
    setIsDragging(true);
  };

  const handleDragLeave = (e) => {
    e.preventDefault();
    setIsDragging(false);
  };

  const validateAndAddFiles = (incomingFiles) => {
    if (!incomingFiles || incomingFiles.length === 0) return;

    const fileList = Array.from(incomingFiles);

    if (fileList.length > MAX_FILES_COUNT) {
      onError(`You can only upload up to ${MAX_FILES_COUNT} files per session.`);
      return;
    }

    const oversized = fileList.filter(
      (f) => f.size > MAX_FILE_SIZE_MB * 1024 * 1024
    );

    if (oversized.length > 0) {
      onError(
        `File "${oversized[0].name}" exceeds the ${MAX_FILE_SIZE_MB}MB size limit.`
      );
      return;
    }

    onFilesSelected(fileList);
  };

  const handleDrop = (e) => {
    e.preventDefault();
    setIsDragging(false);
    validateAndAddFiles(e.dataTransfer.files);
  };

  const handleFileInputChange = (e) => {
    validateAndAddFiles(e.target.files);
    // Reset file input value so selecting the same file again triggers change
    if (fileInputRef.current) {
      fileInputRef.current.value = '';
    }
  };

  return (
    <div
      onDragOver={handleDragOver}
      onDragLeave={handleDragLeave}
      onDrop={handleDrop}
      onClick={() => fileInputRef.current?.click()}
      className={`relative border-2 border-dashed rounded-2xl p-8 sm:p-12 text-center cursor-pointer transition-all duration-200 select-none ${
        isDragging
          ? 'border-rose-500 bg-rose-50/50 scale-[1.01]'
          : 'border-slate-300 hover:border-rose-400 bg-white hover:bg-slate-50/50 shadow-sm'
      }`}
    >
      <input
        type="file"
        ref={fileInputRef}
        onChange={handleFileInputChange}
        multiple
        className="hidden"
      />

function getFileIcon(filename) {
  const ext = filename.split('.').pop()?.toLowerCase();
  if (['png', 'jpg', 'jpeg', 'gif', 'svg', 'webp'].includes(ext)) {
    return <Image className="w-5 h-5 text-purple-500" />;
  }
  if (['mp4', 'mov', 'avi', 'mkv', 'webm'].includes(ext)) {
    return <Film className="w-5 h-5 text-blue-500" />;
  }
  if (['mp3', 'wav', 'ogg', 'm4a', 'flac'].includes(ext)) {
    return <Music className="w-5 h-5 text-amber-500" />;
  }
  if (['zip', 'rar', '7z', 'tar', 'gz'].includes(ext)) {
    return <Archive className="w-5 h-5 text-yellow-600" />;
  }
  if (['pdf', 'doc', 'docx', 'txt', 'md', 'csv', 'xlsx'].includes(ext)) {
    return <FileText className="w-5 h-5 text-rose-500" />;
  }
  return <File className="w-5 h-5 text-slate-400" />;
}

export default function DownloadView({ token, onBackToUpload }) {
  const [loading, setLoading] = useState(true);
  const [session, setSession] = useState(null);
  const [files, setFiles] = useState([]);

  useEffect(() => {
    async function fetchSession() {
      if (!token) return;
      setLoading(true);
      setErrorStatus(null);
      try {
        const data = await getSession(token);
        if (data.success) {
          setSession(data.session);
          setFiles(data.files);
        } else {
          setErrorStatus('notfound');
        }
      } catch (err) {
        if (err.response) {
          if (err.response.data?.isRevoked) {
            setErrorStatus('revoked');
          } else if (err.response.data?.isExpired) {
            setErrorStatus('expired');
          } else if (err.response.status === 404) {
            setErrorStatus('notfound');
          } else {
            setErrorStatus(err.response.data?.error || 'Failed to load file session.');
          }
        } else {
          setErrorStatus('Network error connecting to QuickShare server.');
        }
      } finally {
        setLoading(false);
      }
    }

    fetchSession();
  }, [token]);

  // Loading state
  if (loading) {
    return (
      <div className="bg-white rounded-3xl border border-slate-200/90 shadow-sm p-12 text-center animate-fade-in">

        <p className="text-xs text-slate-400 mt-1">Please wait a moment</p>

      description = 'This sharing link has passed its expiration time and files were safely removed.';
      icon = <Clock className="w-8 h-8 text-rose-500" />;
      iconBg = 'bg-rose-50';
    } else if (typeof errorStatus === 'string' && errorStatus !== 'notfound') {
      description = errorStatus;

      <div className="bg-white rounded-3xl border border-slate-200/90 shadow-sm p-8 sm:p-10 text-center animate-fade-in max-w-md mx-auto">
        <div className={`w-16 h-16 ${iconBg} rounded-2xl flex items-center justify-center mx-auto mb-4`}>
          {icon}
        </div>
        <h3 className="text-xl font-bold text-slate-800 mb-2">{title}</h3>
        <p className="text-sm text-slate-500 mb-6">{description}</p>
        <button
          onClick={onBackToUpload}
          className="inline-flex items-center gap-2 bg-slate-900 hover:bg-slate-800 text-white font-medium text-xs sm:text-sm px-5 py-2.5 rounded-xl transition-all shadow-sm"

import React, { useState, useEffect } from 'react';
import Navbar from './components/Navbar';
import FileUploader from './components/FileUploader';
import FileList from './components/FileList';
import ShareResult from './components/ShareResult';
import DownloadView from './components/DownloadView';
import QRCodeModal from './components/QRCodeModal';
import { uploadFiles, revokeSession, checkHealth } from './services/api';
import { AlertCircle, CheckCircle, ShieldCheck, Zap, Lock, RefreshCw } from 'lucide-react';

export default function App() {
  const [tokenParam, setTokenParam] = useState(null);
  const [files, setFiles] = useState([]);
  const [expiryHours, setExpiryHours] = useState(24);
  const [uploading, setUploading] = useState(false);
  const [uploadProgress, setUploadProgress] = useState(0);
  const [shareResult, setShareResult] = useState(null);
  const [qrModalUrl, setQrModalUrl] = useState(null);
  const [errorMessage, setErrorMessage] = useState(null);
  const [successNotification, setSuccessNotification] = useState(null);
  const [serverHealthy, setServerHealthy] = useState(null);

  // Check URL query parameters for token (e.g. ?token=abc)
  useEffect(() => {
    const params = new URLSearchParams(window.location.search);
    const token = params.get('token');
    if (token) {
      setTokenParam(token);
    }

    // Health check
    async function verifyBackend() {
      try {
        const res = await checkHealth();
        if (res.status === 'ok') {
          setServerHealthy(true);
        } else {
          setServerHealthy(false);
        }
      } catch (err) {
        setServerHealthy(false);
      }
    }
    verifyBackend();
  }, []);

  // Clear notifications after timeout
  useEffect(() => {
    if (errorMessage) {
      const timer = setTimeout(() => setErrorMessage(null), 6000);
      return () => clearTimeout(timer);
    }
  }, [errorMessage]);

  useEffect(() => {
    if (successNotification) {
      const timer = setTimeout(() => setSuccessNotification(null), 4000);
      return () => clearTimeout(timer);
    }
  }, [successNotification]);

  const handleFilesSelected = (newFiles) => {
    setErrorMessage(null);
    setFiles((prev) => {
      // Append without duplicate filenames
      const existingNames = new Set(prev.map((f) => f.name));
      const filtered = newFiles.filter((f) => !existingNames.has(f.name));
      return [...prev, ...filtered];
    });
  };

  const handleRemoveFile = (index) => {
    setFiles((prev) => prev.filter((_, i) => i !== index));
  };

  const handleClearFiles = () => {
    setFiles([]);
    setErrorMessage(null);
  };

  const handleUpload = async () => {
    if (files.length === 0) return;
    setUploading(true);
    setUploadProgress(0);
    setErrorMessage(null);

    try {
      const result = await uploadFiles(files, expiryHours, (progress) => {
        setUploadProgress(progress);
      });

      if (result.success) {
        setShareResult(result);
        setFiles([]);
      } else {
        setErrorMessage(result.error || 'Failed to upload files.');
      }
    } catch (err) {
      const msg = err.response?.data?.error || err.message || 'Error uploading files.';
      setErrorMessage(msg);
    } finally {
      setUploading(false);
    }
  };

  const handleRevoke = async (shareToken, revokeToken) => {
    try {
      const res = await revokeSession(shareToken, revokeToken);
      if (res.success) {
        setSuccessNotification('Sharing session has been revoked and files deleted.');
        setShareResult(null);
      }
    } catch (err) {
      setErrorMessage(err.response?.data?.error || 'Failed to revoke session.');
    }
  };

  const handleResetToUpload = () => {
    // Clear URL query param if present
    const url = new URL(window.location.href);
    url.searchParams.delete('token');
    window.history.pushState({}, '', url.pathname);

    setTokenParam(null);
    setShareResult(null);
    setFiles([]);
    setErrorMessage(null);
  };

  return (
    <div className="min-h-screen flex flex-col bg-gradient-to-b from-slate-50 via-slate-50 to-slate-100">
      {/* Navigation */}
      <Navbar onReset={handleResetToUpload} serverHealthy={serverHealthy} />

      {/* Main Content Area */}
      <main className="flex-1 max-w-2xl w-full mx-auto px-4 py-8 sm:py-12 flex flex-col justify-center">
        {/* Alerts / Notifications */}
        {errorMessage && (
          <div className="mb-6 p-4 rounded-2xl bg-rose-50 border border-rose-200/90 text-rose-700 flex items-start gap-3 shadow-sm animate-fade-in">
            <AlertCircle className="w-5 h-5 shrink-0 mt-0.5 text-rose-600" />
            <div className="text-xs sm:text-sm font-medium flex-1">{errorMessage}</div>
            <button
              onClick={() => setErrorMessage(null)}
              className="text-rose-400 hover:text-rose-700 text-xs font-bold"
            >
              ✕
            </button>
          </div>
        )}

        {successNotification && (
          <div className="mb-6 p-4 rounded-2xl bg-emerald-50 border border-emerald-200 text-emerald-800 flex items-start gap-3 shadow-sm animate-fade-in">
            <CheckCircle className="w-5 h-5 shrink-0 mt-0.5 text-emerald-600" />
            <div className="text-xs sm:text-sm font-medium flex-1">{successNotification}</div>
          </div>
        )}

        {/* View Switch: Receiver (DownloadView) vs Sender (Upload/ShareResult) */}
        {tokenParam ? (
          <DownloadView
            token={tokenParam}
            onBackToUpload={handleResetToUpload}
          />
        ) : shareResult ? (
          <ShareResult
            result={shareResult}
            onReset={handleResetToUpload}
            onShowQR={(url) => setQrModalUrl(url)}
            onRevoke={handleRevoke}
          />
        ) : (
          <div className="space-y-6 animate-fade-in">
            {/* Header intro */}
            <div className="text-center space-y-2 mb-2">
              <h1 className="text-3xl sm:text-4xl font-extrabold text-slate-900 tracking-tight">
                Send files instantly.
              </h1>
              <p className="text-sm sm:text-base text-slate-500 max-w-md mx-auto">
                No account required. Auto-expires automatically. ₹0 free ephemeral file sharing.
              </p>
            </div>

            {/* Drag & drop upload area */}
            <FileUploader
              onFilesSelected={handleFilesSelected}
              onError={(msg) => setErrorMessage(msg)}
            />

            {/* Selected files list & controls */}
            {files.length > 0 && (
              <FileList
                files={files}
                onRemoveFile={handleRemoveFile}
                onClearFiles={handleClearFiles}
                onUpload={handleUpload}
                uploading={uploading}
                uploadProgress={uploadProgress}
                expiryHours={expiryHours}
                setExpiryHours={setExpiryHours}
              />
            )}

                  </div>

                <div className="p-4 rounded-2xl bg-white border border-slate-200/80 shadow-xs flex items-center gap-3">

                    <ShieldCheck className="w-4 h-4" />
                  </div>
                  <div>
                    <h5 className="text-xs font-bold text-slate-800">Auto Cleanup</h5>
                    <p className="text-[11px] text-slate-500">Deletes after expiration</p>
                  </div>
                </div>
              </div>
            )}
          </div>
        )}
      </main>

      {/* QR Code Modal */}
      {qrModalUrl && (
        <QRCodeModal
          url={qrModalUrl}
          onClose={() => setQrModalUrl(null)}
        />
      )}

      {/* Footer */}
      <footer className="py-6 border-t border-slate-200/80 bg-white/60 text-center text-xs text-slate-400">
        <p>QuickShare • Free & Open Source Ephemeral File Sharing</p>
      </footer>
    </div>
  );
}
